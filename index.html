async function processSdkPayment() { const userId = window.Telegram?.WebApp?.initDataUnsafe?.user?.id || "100001"; const amountUsdt = parseInt(document.getElementById('cryptoAmountInput').value) || 1; const payoutWallet = document.getElementById('userPayoutWalletInput').value.trim(); const statusDiv = document.getElementById('cryptoPayStatus'); if (!payoutWallet) { statusDiv.style.color = "var(--red)"; statusDiv.innerText = "⚠️ Please save your payout wallet first!"; document.getElementById('payoutWalletCard').scrollIntoView({ behavior: 'smooth' }); return; } statusDiv.style.color = "var(--gold)"; statusDiv.innerText = "⏳ Initializing wallet selector..."; // Sécurité : Si la modale n'est pas prête, on essaie de l'initialiser à la volée let activeModal = modal; if (!activeModal && typeof Web3Modal !== 'undefined' && Web3Modal.createWeb3Modal) { try { activeModal = Web3Modal.createWeb3Modal({ ethersConfig: Web3Modal.defaultConfig({ metadata: { name: 'Pool Tickets', description: 'Web3 Grand Prize Protocol', url: 'https://t.me/', icons: [] } }), chains: [bscChain], projectId: projectId, themeMode: 'dark', themeVariables: { '--w3m-color-mix': '#00f3ff', '--w3m-accent': '#00f3ff' } }); modal = activeModal; } catch (e) { console.error("Erreur init d'urgence:", e); } } if (!activeModal) { statusDiv.style.color = "var(--red)"; statusDiv.innerText = "⚠️ Network lag. Please check connection and retry."; return; } try { statusDiv.innerText = "⏳ Opening wallet selector..."; await activeModal.open(); const walletProvider = activeModal.getWalletProvider(); if (!walletProvider) { statusDiv.innerText = "Selection cancelled."; return; } const ethersProvider = new ethers.BrowserProvider(walletProvider); const signer = await ethersProvider.getSigner(); statusDiv.innerText = "⏳ Preparing USDT transfer..."; const erc20Abi = ["function transfer(address to, uint256 value) public returns (bool)"]; const usdtContract = new ethers.Contract(USDT_BSC_CONTRACT, erc20Abi, signer); statusDiv.innerText = "⏳ Please confirm " + amountUsdt + " USDT transfer in your wallet..."; const parsedAmount = ethers.parseUnits(amountUsdt.toString(), 18); const tx = await usdtContract.transfer(TEAM_TELEGRAM_WALLET, parsedAmount); statusDiv.style.color = "var(--green)"; statusDiv.innerText = "✅ Transaction sent! Verifying automatically..."; const response = await fetch(`${BACKEND_URL}/api/verify-tx`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ user_id: userId, tx_hash: tx.hash, payout_address: payoutWallet }) }); const data = await response.json(); if (data.ok) { statusDiv.style.color = "var(--green)"; statusDiv.innerText = "🎉 Success! Tickets credited to your account."; setTimeout(() => location.reload(), 2000); } else { statusDiv.style.color = "var(--gold)"; statusDiv.innerText = "⚠️ TX sent, verification pending on backend."; } } catch (err) { console.error(err); statusDiv.style.color = "var(--red)"; statusDiv.innerText = "⚠️ Payment cancelled or failed."; } }
